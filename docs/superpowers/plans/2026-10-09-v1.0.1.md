```ts
/**
 * Пул ходов для «ошибки» уровня: hard берёт только сильнейший из неточных ходов
 * (случайный среди равных), остальные уровни — случайный среди всех неточных.
 */
export function mistakePool(
  scored: readonly { cell: number; score: number }[],
  bestScore: number,
  difficulty: Difficulty,
): { cell: number; score: number }[] {
  const others = scored.filter((s) => s.score !== bestScore);
  if (difficulty !== 'hard') return others;
  let max = -Infinity;
  for (const s of others) if (s.score > max) max = s.score;
  return others.filter((s) => s.score === max);
}
```

## Task 4: Canvas учитывает prefers-reduced-motion

**Контекст (зачем):** P2/P3 анализа: CSS-анимации уважают системную настройку «уменьшить движение», а canvas-анимации (появление знака, победная линия) — нет (BACKLOG, строка 25). По спецификации: `matchMedia` читается **на каждый старт анимации** (не кэшируем — предпочтение можно менять на лету); при включённой настройке рисуется финальное состояние сразу, без rAF; звук, статистика и результаты не зависят от настройки. Unit-тестов нет: `main.ts` не тестируется (Vitest в node-окружении без matchMedia) — проверка живая.

### Шаг 1 — `src/main.ts`

1. Добавить функцию перед `beginMarkAnim` (строка 279):

```ts
/**
 * Системная настройка «уменьшить движение». Читается на каждый старт анимации:
 * предпочтение можно менять на лету, пока игра открыта.
 */
function prefersReducedMotion(): boolean {
  try {
    return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  } catch {
    return false;
  }
}
```

2. `beginMarkAnim` / `beginWinAnim` (строки 279–287) — добавить ранний выход:

```ts
function beginMarkAnim(cell: number): void {
  if (prefersReducedMotion()) return; // знак рисуется сразу целиком
  markAnim = { cell, start: performance.now() };
  runAnim();
}

function beginWinAnim(): void {
  if (prefersReducedMotion()) return; // линия рисуется сразу целиком
  winAnim = { start: performance.now() };
  runAnim();
}
```

**Почему это работает:** все вызывающие (`aiMove`, клик, клавиатура, `finishGameIfNeeded`) сразу после `begin*Anim` зовут `draw()`; при `markAnim`/`winAnim === null` рисуется финальное состояние (`winProgress` по умолчанию 1, знак целиком). `clearAnims` не трогаем.

```ts
/**
 * Геометрия стилизованного знака в точке (cx, cy) поля стороной size.
 * Набор 1 — знаки «от руки» (X изогнут, O — одно кольцо); Набор 2 — руны
 * (X прямой с засечками, O — двойное кольцо).
 */
export function markStrokes(
  mark: Mark,
  cx: number,
  cy: number,
  size: number,
  set: MarkSet = 'set-1',
): MarkStroke[] {
  // half-cell * 0.52: знак занимает ~52% клетки и не касается линий сетки
  const h = (size / 3) * 0.26;
  if (mark === 'X') {
    if (set === 'set-2') {
      // Руна: две прямые диагонали + короткие засечки на концах.
      const s = h * 0.3;
      const main: MarkStroke[] = [
        { kind: 'quadratic', x1: cx - h, y1: cy - h, cx, cy, x2: cx + h, y2: cy + h },
        { kind: 'quadratic', x1: cx + h, y1: cy - h, cx, cy, x2: cx - h, y2: cy + h },
      ];
      const serifs: MarkStroke[] = [
        { kind: 'quadratic', x1: cx - h - s, y1: cy - h + s, cx, cy, x2: cx - h + s, y2: cy - h - s },
        { kind: 'quadratic', x1: cx + h - s, y1: cy + h + s, cx, cy, x2: cx + h + s, y2: cy + h - s },
        { kind: 'quadratic', x1: cx + h - s, y1: cy - h - s, cx, cy, x2: cx + h + s, y2: cy - h + s },
        { kind: 'quadratic', x1: cx - h + s, y1: cy + h + s, cx, cy, x2: cx - h - s, y2: cy + h - s },
      ];
      return [...main, ...serifs];
    }
    // изгиб перпендикулярен направлению штриха; оба штриха — с одинаковым поворотом
    const q = (h * 0.16) / Math.SQRT2;
    return [
      { kind: 'quadratic', x1: cx - h, y1: cy - h, cx: cx - q, cy: cy + q, x2: cx + h, y2: cy + h },
      { kind: 'quadratic', x1: cx + h, y1: cy - h, cx: cx - q, cy: cy - q, x2: cx - h, y2: cy + h },
    ];
  }
  const from = -Math.PI / 2;
  if (set === 'set-2') {
    // Двойное кольцо: наружное как в Наборе 1, внутреннее уже.
    return [
      { kind: 'arc', cx, cy, r: h, from, to: from + Math.PI * 2 },
      { kind: 'arc', cx, cy, r: h * 0.62, from, to: from + Math.PI * 2 },
    ];
  }
  return [{ kind: 'arc', cx, cy, r: h, from, to: from + Math.PI * 2 }];
}
```
